// ── Desk mode: compositing engine ────────────────────────────────────────────
// The case never moves: it stays at the origin exactly as in the normal
// viewer, so placement, picking, outlines and every animation keep working
// untouched. Instead the photo's camera, its lights and the shadow catcher are
// expressed relative to the case (desk space -> model space), which is the
// same picture as standing the case on the desk. Per frame this costs one
// textured quad (the photo) + a few flat shadow catchers on top of the case
// itself; the shadow textures are only re-baked while something moves.
const DESK_SHADOW_EXTENT = 1.15 // half-size of the baked shadow area, model units
const DESK_GRAIN = 0.22 // film grain over the whole picture when a view doesn't set its own (0 = off)

function deskImageSrc(view: DeskView): string {
    if (typeof window === "undefined") return view.image.src
    const px = Math.max(window.innerWidth, window.innerHeight) * Math.min(window.devicePixelRatio || 1, 2)
    return px <= 1500 && view.image.srcSmall ? view.image.srcSmall : view.image.src
}
function deskResolveView(view: DeskView, portrait: boolean): DeskView {
    return portrait && view.portrait ? ({ ...view, ...view.portrait, portrait: undefined } as DeskView) : view
}
// the lights of a view as a normal lighting preset (so the usual cross-fade code drives them)
function deskPreset(base: EnvPreset, view: DeskView): EnvPreset {
    const l = view.lighting
    return { ...base, id: "desk", effect: "none", ambient: l.ambient, key: l.key, fill: l.fill, rim: l.rim, instant: true } as EnvPreset
}
function deskYaw(d: any): number { return ((d.view?.product.rotationY ?? 0) * Math.PI) / 180 + d.yaw }

function deskEnsure(world: any): any {
    if (world.desk) return world.desk
    const { THREE } = world
    const coarse = typeof window !== "undefined" && !!window.matchMedia?.("(pointer: coarse)").matches
    const rt = (n: number, depth = false) => { const t = new THREE.WebGLRenderTarget(n, n, { depthBuffer: depth }); t.texture.generateMipmaps = false; return t }
    const d: any = {
        active: false, view: null, yaw: 0, dirtyUntil: 0, tick: 0, coarse, textures: new Map<string, any>(),
        scene: new THREE.Scene(), recvScene: new THREE.Scene(), tmp: new THREE.Vector3(), base: new THREE.Vector3(), tmpColor: new THREE.Color(),
        rawDir: rt(512, true), tight: rt(512), tmpA: rt(512), wide: rt(256), tmpB: rt(256), rawAO: rt(256, true), ao: rt(256), tmpC: rt(256), glow: rt(256),
        lightCam: new THREE.OrthographicCamera(-1, 1, 1, -1, 0.01, 10),
        bakeCam: new THREE.OrthographicCamera(-DESK_SHADOW_EXTENT, DESK_SHADOW_EXTENT, DESK_SHADOW_EXTENT, -DESK_SHADOW_EXTENT, 0.01, 100),
        quadCam: new THREE.OrthographicCamera(-1, 1, 1, -1, 0, 1), quadScene: new THREE.Scene(),
    }
    // the photograph, cropped to the viewer ("cover")
    d.bgMat = new THREE.ShaderMaterial({
        depthTest: false, depthWrite: false,
        uniforms: { tMap: { value: null }, uCrop: { value: new THREE.Vector4(0, 0, 1, 1) } },
        vertexShader: "uniform vec4 uCrop; varying vec2 vUv; void main(){ vUv = uCrop.xy + uv * uCrop.zw; gl_Position = vec4(position.xy, 1.0, 1.0); }",
        fragmentShader: "uniform sampler2D tMap; varying vec2 vUv; void main(){ gl_FragColor = vec4(texture2D(tMap, vUv).rgb, 1.0);\n#include <colorspace_fragment>\n}",
    })
    const bg = new THREE.Mesh(new THREE.PlaneGeometry(2, 2), d.bgMat)
    bg.frustumCulled = false; bg.renderOrder = -2
    // Shadow catchers: invisible stand-ins for the surfaces in the photo (desk
    // top, a shelf, the wall behind...). Each one looks its own shadow up in a
    // map of the case seen from the light, so the shadow runs across a surface,
    // drops off its edge onto the one below and climbs whatever stands behind.
    d.shadowMat = new THREE.ShaderMaterial({
        transparent: true, depthWrite: false, depthFunc: THREE.LessEqualDepth, polygonOffset: true, polygonOffsetFactor: -1, polygonOffsetUnits: -2,
        uniforms: {
            tTight: { value: d.tight.texture }, tWide: { value: d.wide.texture }, tAO: { value: d.ao.texture }, tGlow: { value: d.glow.texture },
            uLight: { value: new THREE.Matrix4() }, uRange: { value: 1 }, uMaxLen: { value: 1 }, uSoftLen: { value: 0.5 },
            uBase: { value: new THREE.Vector3() }, uExtent: { value: DESK_SHADOW_EXTENT }, uContactBand: { value: 0.012 },
            uColor: { value: new THREE.Color(0x000000) }, uOpacity: { value: 0.5 }, uContact: { value: 0.6 },
            uTint: { value: new THREE.Color(0x888888) }, uGlow: { value: 0 },
        },
        vertexShader: "varying vec3 vW; void main(){ vec4 w = modelMatrix * vec4(position, 1.0); vW = w.xyz; gl_Position = projectionMatrix * viewMatrix * w; }",
        fragmentShader: [
            "uniform sampler2D tTight; uniform sampler2D tWide; uniform sampler2D tAO; uniform sampler2D tGlow; uniform mat4 uLight; uniform float uRange; uniform float uMaxLen; uniform float uSoftLen;",
            "uniform vec3 uBase; uniform float uExtent; uniform float uContactBand; uniform vec3 uColor; uniform float uOpacity; uniform float uContact; uniform vec3 uTint; uniform float uGlow; varying vec3 vW;",
            "void main(){",
            // cast shadow: r = how much of the light the case blocks here, g = r * how far along the light the blocker is
            "  vec4 lp = uLight * vec4(vW, 1.0);",
            "  vec4 a = texture2D(tTight, lp.xy); vec4 b = texture2D(tWide, lp.xy);",
            "  float dist = (lp.z - b.g / max(b.r, 0.001)) * uRange;", // how far behind the blocker this surface is
            "  float s = mix(a.r, b.r, smoothstep(0.0, uSoftLen, dist));", // crisp near the case, spreading with distance
            "  s *= smoothstep(-0.01, 0.03, dist) * (1.0 - smoothstep(uMaxLen * 0.55, uMaxLen, dist));",
            "  if (lp.x < 0.0 || lp.x > 1.0 || lp.y < 0.0 || lp.y > 1.0) s = 0.0;",
            // contact shadow + bounce glow: only on the surface the case stands on
            "  float onBase = 1.0 - smoothstep(uContactBand * 0.4, uContactBand, abs(vW.y - uBase.y));",
            "  vec2 buv = vec2((vW.x - uBase.x) / (2.0 * uExtent) + 0.5, (uBase.z - vW.z) / (2.0 * uExtent) + 0.5);",
            "  float ao = texture2D(tAO, buv).r * onBase;",
            "  float al = 1.0 - (1.0 - uOpacity * s) * (1.0 - uContact * ao * ao);",
            "  float g = uGlow * smoothstep(0.0, 0.55, texture2D(tGlow, buv).r) * onBase * (1.0 - 0.85 * s) * (1.0 - 0.6 * ao) * (1.0 - al);",
            "  float outA = al + g; if (outA < 0.003) discard;",
            "  gl_FragColor = vec4((uColor * al + uTint * g) / outA, outA);",
            "  #include <colorspace_fragment>",
            "}",
        ].join("\n"),
    })
    d.depthMat = new THREE.MeshBasicMaterial({ colorWrite: false })
    d.recv = new THREE.Group() // desk space -> model space; holds one unit cube per catcher
    d.recvGeo = new THREE.BoxGeometry(1, 1, 1)
    d.recvScene.add(d.recv)
    // calibration helpers: 10 cm grid on the standing surface + the catchers' outlines, in desk space
    d.frame = new THREE.Group()
    const grid = new THREE.GridHelper(2, 20, 0xff3b30, 0x00c2ff)
    grid.material.depthTest = false; grid.material.transparent = true; grid.material.opacity = 0.8
    d.wires = new THREE.Group()
    d.frame.add(grid, new THREE.AxesHelper(0.3), d.wires); d.frame.visible = false
    d.scene.add(bg, d.frame)
    // the case seen from the light: r = covered, g = depth along the light
    d.lightMat = new THREE.ShaderMaterial({
        side: THREE.DoubleSide,
        vertexShader: [
            "void main(){",
            "  #ifdef USE_INSTANCING",
            "    gl_Position = projectionMatrix * viewMatrix * modelMatrix * instanceMatrix * vec4(position, 1.0);",
            "  #else",
            "    gl_Position = projectionMatrix * viewMatrix * modelMatrix * vec4(position, 1.0);",
            "  #endif",
            "}",
        ].join("\n"),
        fragmentShader: "void main(){ gl_FragColor = vec4(1.0, gl_FragCoord.z, 0.0, 1.0); }",
    })
    // the case flattened straight down onto what it stands on (contact shadow): r = closeness to the surface
    d.projMat = new THREE.ShaderMaterial({
        side: THREE.DoubleSide,
        uniforms: { uY0: { value: 0 }, uHMax: { value: 1 } },
        vertexShader: [
            "uniform float uY0; uniform float uHMax; varying float vH;",
            "void main(){",
            "  #ifdef USE_INSTANCING",
            "    vec4 w = modelMatrix * instanceMatrix * vec4(position, 1.0);",
            "  #else",
            "    vec4 w = modelMatrix * vec4(position, 1.0);",
            "  #endif",
            "  vH = clamp(max(w.y - uY0, 0.0) / uHMax, 0.0, 1.0);",
            "  gl_Position = projectionMatrix * viewMatrix * vec4(w.x, uY0, w.z, 1.0);",
            "  gl_Position.z = (vH * 1.98 - 0.99) * gl_Position.w;", // lowest part wins
            "}",
        ].join("\n"),
        fragmentShader: "varying float vH; void main(){ gl_FragColor = vec4(1.0 - vH, 0.0, 0.0, 1.0); }",
    })
    d.blurMat = new THREE.ShaderMaterial({
        depthTest: false, depthWrite: false,
        uniforms: { tMap: { value: null }, uStep: { value: new THREE.Vector2() } },
        vertexShader: "varying vec2 vUv; void main(){ vUv = uv; gl_Position = vec4(position.xy, 0.0, 1.0); }",
        fragmentShader: [
            "uniform sampler2D tMap; uniform vec2 uStep; varying vec2 vUv;",
            "void main(){",
            "  vec4 s = texture2D(tMap, vUv) * 0.1964825501511404;",
            "  s += (texture2D(tMap, vUv + uStep * 1.411764705882353) + texture2D(tMap, vUv - uStep * 1.411764705882353)) * 0.2969069646728344;",
            "  s += (texture2D(tMap, vUv + uStep * 3.2941176470588234) + texture2D(tMap, vUv - uStep * 3.2941176470588234)) * 0.09447039785044732;",
            "  s += (texture2D(tMap, vUv + uStep * 5.176470588235294) + texture2D(tMap, vUv - uStep * 5.176470588235294)) * 0.010381362401148057;",
            "  gl_FragColor = s;",
            "}",
        ].join("\n"),
    })
    d.quadScene.add(new THREE.Mesh(new THREE.PlaneGeometry(2, 2), d.blurMat))
    // Film grain over photo + case together, so they read as one exposure. Not
    // an overlay: the finished frame is copied once and re-developed -- grain
    // is noise in the *exposure* (multiplied in linear light), strongest in the
    // mid-tones, nearly gone in clear highlights, with a little base fog in the
    // blacks; it comes from a small pre-blurred tile (soft clumps, mostly
    // luminance, a trace of colour) that jumps 24 times a second like frames.
    const gN = 256, gData = new Uint8Array(gN * gN * 4), gRaw = new Float32Array(gN * gN * 3)
    for (let i = 0; i < gRaw.length; i++) gRaw[i] = (Math.random() + Math.random() + Math.random() + Math.random() - 2) // ~gaussian
    for (let y = 0; y < gN; y++) for (let x = 0; x < gN; x++) for (let c = 0; c < 3; c++) {
        let v = 0
        for (let dy = -1; dy <= 1; dy++) for (let dx = -1; dx <= 1; dx++) v += gRaw[(((y + dy + gN) % gN) * gN + ((x + dx + gN) % gN)) * 3 + c] * (dx === 0 && dy === 0 ? 4 : dx === 0 || dy === 0 ? 2 : 1)
        gData[(y * gN + x) * 4 + c] = Math.max(0, Math.min(255, 127.5 + (v / 16) * 150))
    }
    for (let i = 3; i < gData.length; i += 4) gData[i] = 255
    d.grainTex = new THREE.DataTexture(gData, gN, gN)
    d.grainTex.wrapS = d.grainTex.wrapT = THREE.RepeatWrapping; d.grainTex.minFilter = d.grainTex.magFilter = THREE.LinearFilter; d.grainTex.needsUpdate = true
    d.grainMat = new THREE.ShaderMaterial({
        depthTest: false, depthWrite: false,
        uniforms: { tImg: { value: null }, tGrain: { value: d.grainTex }, uOff: { value: new THREE.Vector4() }, uAmt: { value: 0 }, uSize: { value: 1 } },
        vertexShader: "varying vec2 vUv; void main(){ vUv = uv; gl_Position = vec4(position.xy, 0.0, 1.0); }",
        fragmentShader: [
            "uniform sampler2D tImg; uniform sampler2D tGrain; uniform vec4 uOff; uniform float uAmt; uniform float uSize; varying vec2 vUv;",
            "void main(){",
            "  vec3 c = texture2D(tImg, vUv).rgb;",
            "  vec2 p = gl_FragCoord.xy / (256.0 * uSize);",
            "  vec3 g = (texture2D(tGrain, p + uOff.xy).rgb - 0.5) * 0.85 + (texture2D(tGrain, p * 0.61 + uOff.zw).gbr - 0.5) * 0.3;", // fine grain + larger clumps
            "  g = mix(vec3(dot(g, vec3(0.3333))), g, 0.3);", // mostly luminance, a trace of colour
            "  vec3 lin = c * c;", // display -> (about) linear light
            "  float l = dot(lin, vec3(0.2126, 0.7152, 0.0722));",
            "  float resp = 0.35 + 2.2 * sqrt(l) * (1.0 - 0.8 * l);", // film response: mid-tones grainiest
            "  lin *= exp2(g * uAmt * resp * 2.0);", // noise in exposure
            "  lin += g * uAmt * 0.02;", // base fog in the blacks
            "  gl_FragColor = vec4(sqrt(max(lin, 0.0)), 1.0);",
            "}",
        ].join("\n"),
    })
    d.grainScene = new THREE.Scene()
    d.grainScene.add(new THREE.Mesh(new THREE.PlaneGeometry(2, 2), d.grainMat))
    d.zero2 = new THREE.Vector2(); d.bufSize = new THREE.Vector2()
    // light bounced off the wall and the desk top
    d.bounce = new THREE.HemisphereLight(0xffffff, 0xffffff, 0)
    world.scene.add(d.bounce)
    world.desk = d
    return d
}

// (Re)build the shadow catchers of a view: boxes [minX, minY, minZ, maxX, maxY, maxZ] in desk space.
function deskSetReceivers(world: any, view: DeskView): void {
    const d = world.desk, { THREE } = world
    const boxes = view.receivers && view.receivers.length ? view.receivers : [[-3, -0.05, -3, 3, 0, 3]]
    const key = JSON.stringify(boxes)
    if (d.recvKey === key) return
    d.recvKey = key
    d.recv.clear()
    for (const w of d.wires.children) w.geometry.dispose()
    d.wires.clear()
    for (const b of boxes) {
        const m = new THREE.Mesh(d.recvGeo, d.shadowMat)
        m.position.set((b[0] + b[3]) / 2, (b[1] + b[4]) / 2, (b[2] + b[5]) / 2)
        m.scale.set(Math.max(1e-4, b[3] - b[0]), Math.max(1e-4, b[4] - b[1]), Math.max(1e-4, b[5] - b[2]))
        m.frustumCulled = false
        d.recv.add(m)
        const wire = new THREE.LineSegments(new THREE.EdgesGeometry(d.recvGeo), new THREE.LineBasicMaterial({ color: 0xffd400, depthTest: false, transparent: true }))
        wire.position.copy(m.position); wire.scale.copy(m.scale)
        d.wires.add(wire)
    }
}

function deskLoadTexture(world: any, src: string): Promise<any> {
    const d = deskEnsure(world), { THREE } = world
    const hit = d.textures.get(src)
    if (hit) return Promise.resolve(hit)
    return new Promise((resolve, reject) => {
        const loader = new THREE.TextureLoader(); loader.setCrossOrigin("anonymous")
        loader.load(src, (tex: any) => {
            tex.colorSpace = THREE.SRGBColorSpace; tex.generateMipmaps = false
            tex.minFilter = THREE.LinearFilter; tex.magFilter = THREE.LinearFilter
            tex.wrapS = THREE.MirroredRepeatWrapping; tex.repeat.set(2, 1) // only the reflection dome uses this: the photo, mirrored round the back
            if (world.disposed) { tex.dispose(); reject(new Error("disposed")); return }
            d.textures.set(src, tex); resolve(tex)
        }, undefined, reject)
    })
}

// Image-based light for the case, assembled from the view itself: the photo
// wrapped round the case as its surroundings (centred behind the case as the
// camera sees it), a bright panel where the key light is, a soft panel for
// the fill, and the standing surface's colour underneath. Aluminium parts
// mirror it; with lighting.ibl set, every other material is lit by it too.
function deskBuildEnv(world: any, tex: any): void {
    const d = world.desk, { THREE, renderer, camera } = world
    const v: DeskView = d.view, ibl = v.lighting.ibl ?? 0
    if (!ibl && !world.studioEnv && !d.envSaved) return // nothing would use it
    const yaw = deskYaw(d)
    const envKey = tex.uuid + "|" + JSON.stringify(v.lighting) + "|" + JSON.stringify(v.camera) + "|" + yaw.toFixed(2)
    if (d.envKey === envKey) return
    d.envKey = envKey
    const up = new THREE.Vector3(0, 1, 0)
    const scene = new THREE.Scene()
    const gain = ibl || 1.5
    const dome = new THREE.Mesh(new THREE.SphereGeometry(5, 32, 16), new THREE.MeshBasicMaterial({ map: tex, side: THREE.BackSide, color: new THREE.Color(gain, gain, gain) }))
    // the photo's centre goes behind the case, away from the camera
    const away = d.tmp.copy(d.base).sub(camera.position); away.y = 0
    dome.rotation.y = Math.atan2(-away.z, away.x)
    scene.add(dome)
    const panel = (l: DeskLight, size: number, boost: number) => {
        const m = new THREE.Mesh(new THREE.PlaneGeometry(size, size), new THREE.MeshBasicMaterial({ color: new THREE.Color(l.color).multiplyScalar(boost * l.intensity), side: THREE.DoubleSide }))
        m.position.set(l.position[0], l.position[1], l.position[2]).normalize().multiplyScalar(4.5).applyAxisAngle(up, -yaw)
        m.lookAt(0, 0, 0); scene.add(m)
    }
    panel(v.lighting.key, 3.2, 1.6)
    if (ibl) {
        panel(v.lighting.fill, 4.5, 0.7)
        const b = v.lighting.bounce
        if (b) {
            const ground = new THREE.Mesh(new THREE.CircleGeometry(4.9, 32), new THREE.MeshBasicMaterial({ color: new THREE.Color(b.ground).multiplyScalar(0.8 * gain), side: THREE.DoubleSide }))
            ground.rotation.x = -Math.PI / 2; ground.position.y = -1.2; scene.add(ground)
        }
    }
    const pmrem = new THREE.PMREMGenerator(renderer)
    const env = pmrem.fromScene(scene, 0.04).texture
    pmrem.dispose()
    scene.traverse((c: any) => { if (c.isMesh) { c.geometry.dispose(); c.material.dispose() } })
    deskSwapEnv(world, env)
    world.scene.environment = ibl ? env : null
}
function deskSwapEnv(world: any, env: any): void {
    const d = world.desk, old = world.studioEnv
    if (!d.envSaved) d.envSaved = old || "none"
    world.studioEnv = env
    world.scene.traverse((c: any) => { if (c.isMesh && c.material && c.material.envMap === old && old) { c.material.envMap = env; c.material.needsUpdate = true } })
    if (old && old !== d.envSaved && old !== env) old.dispose()
}

// Show a view whose photograph is already loaded.
function deskApply(world: any, view: DeskView, tex: any): void {
    const d = deskEnsure(world), { camera, controls } = world
    if (!d.active) {
        d.saved = { pos: camera.position.clone(), fov: camera.fov, enabled: controls.enabled }
        controls.enabled = false
        world.cameraAnim = null
        d.active = true
    }
    if (!d.view || d.view.id !== view.id || d.view.image.src !== view.image.src) d.yaw = 0
    d.view = view
    d.bgMat.uniforms.tMap.value = tex
    d.shadowMat.uniforms.uColor.value.set(view.shadow.color)
    d.shadowMat.uniforms.uOpacity.value = view.shadow.opacity
    d.shadowMat.uniforms.uContact.value = view.shadow.contact
    d.shadowMat.uniforms.uGlow.value = view.shadow.glow ?? 0
    deskSetReceivers(world, view)
    const b = view.lighting.bounce
    d.bounce.intensity = b ? b.intensity : 0
    if (b) { d.bounce.color.set(b.sky); d.bounce.groundColor.set(b.ground) }
    d.frame.visible = !!view.debugGrid
    d.grainMat.uniforms.uAmt.value = view.grain ?? DESK_GRAIN
    // the case shadows itself from the key light (map refreshed with the baked shadows, not every frame)
    const { renderer, lights, THREE } = world
    renderer.shadowMap.enabled = true; renderer.shadowMap.autoUpdate = false
    lights.key.castShadow = true
    const kp = view.lighting.key.position, kl = Math.hypot(kp[0], kp[1], kp[2]), sc = lights.key.shadow.camera
    sc.left = -0.52; sc.right = 0.52; sc.top = 0.52; sc.bottom = -0.52; sc.near = Math.max(0.01, kl - 0.62); sc.far = kl + 0.62
    sc.updateProjectionMatrix()
    const ms = d.coarse ? 1024 : 2048
    if (lights.key.shadow.mapSize.x !== ms) { lights.key.shadow.mapSize.set(ms, ms); lights.key.shadow.map?.dispose(); lights.key.shadow.map = null }
    lights.key.shadow.bias = -0.0006; lights.key.shadow.normalBias = 0.004
    // grade: lifts the case's blacks towards the photo's own darkest tone (as constant-density fog)
    const gr = view.grade
    if (gr && gr.amount > 0) { if (!world.scene.fog) world.scene.fog = new THREE.Fog(gr.color, 0, 1); world.scene.fog.color.set(gr.color) } else world.scene.fog = null
    // keep only this photograph on the GPU
    for (const [src, t] of d.textures) if (t !== tex) { t.dispose(); d.textures.delete(src) }
    d.tex = tex
    deskCamera(world)
    deskBuildEnv(world, tex)
    deskBake(world) // the new view's shadow is there on its first frame
    d.settle = 2; d.dirtyUntil = performance.now() + 1500
}

function deskExit(world: any): void {
    const d = world.desk; if (!d || !d.active) return
    const { camera, controls, renderer } = world
    d.active = false
    camera.clearViewOffset()
    camera.up.set(0, 1, 0)
    camera.fov = d.saved.fov; camera.position.copy(d.saved.pos); camera.lookAt(0, 0, 0)
    const el = renderer.domElement, aspect = (el.clientWidth || 1) / (el.clientHeight || 1)
    camera.aspect = aspect; camera.zoom = fitZoom(aspect); camera.updateProjectionMatrix()
    controls.enabled = d.saved.enabled
    world.lightRig.rotation.y = 0
    d.bounce.intensity = 0
    if (d.envSaved) {
        const env = world.studioEnv
        if (d.envSaved === "none") { world.studioEnv = null; env?.dispose() } else deskSwapEnv(world, d.envSaved)
        d.envSaved = null
    }
    world.scene.environment = null
    world.scene.fog = null
    renderer.shadowMap.enabled = false; renderer.shadowMap.autoUpdate = true
    world.lights.key.castShadow = false
    d.envKey = null; d.tex = null
    for (const [, t] of d.textures) t.dispose()
    d.textures.clear(); d.bgMat.uniforms.tMap.value = null; d.view = null
}

// Every frame: put the camera where the photographer stood (relative to the
// case) and crop its frustum exactly like the photo is cropped.
function deskCamera(world: any): void {
    const d = world.desk, v: DeskView = d.view, box = world.caseBox
    if (!v || !box) return
    const { camera, renderer } = world
    const m = world.rootScale / (v.product.scale || 1) // model units per metre
    const yaw = deskYaw(d), cos = Math.cos(yaw), sin = Math.sin(yaw)
    const bx = (box.min.x + box.max.x) / 2, bz = (box.min.z + box.max.z) / 2
    const toModel = (p: DeskV3, out: any) => {
        const x = p[0] - v.product.position[0], z = p[2] - v.product.position[1]
        return out.set((x * cos - z * sin) * m + bx, p[1] * m + box.min.y, (x * sin + z * cos) * m + bz)
    }
    d.base.set(bx, box.min.y, bz)
    toModel(v.camera.position, camera.position)
    camera.up.set(0, 1, 0)
    camera.lookAt(toModel(v.camera.target, d.tmp))
    if (v.camera.roll) camera.rotateZ((v.camera.roll * Math.PI) / 180)
    const el = renderer.domElement, cw = el.clientWidth || 1, ch = el.clientHeight || 1
    const W = v.image.width, H = v.image.height, av = cw / ch
    let w = W, h = H
    if (av >= W / H) h = W / av; else w = H * av
    const z = Math.max(1, (cw < 760 ? v.framing?.narrowZoom ?? v.framing?.zoom : v.framing?.zoom) ?? 1)
    w /= z; h /= z
    const f = v.framing?.focus ?? [0.5, 0.5]
    const x = Math.min(Math.max(f[0] * W - w / 2, 0), W - w), y = Math.min(Math.max(f[1] * H - h / 2, 0), H - h)
    camera.fov = v.camera.fov; camera.zoom = 1
    camera.setViewOffset(W, H, x, y, w, h)
    d.bgMat.uniforms.uCrop.value.set(x / W, 1 - (y + h) / H, w / W, h / H)
    world.lightRig.rotation.y = -yaw
    const fog = world.scene.fog
    if (fog && v.grade) { // same density over the whole case: one huge ramp, the case sitting at the wanted point of it
        const k = Math.min(0.6, Math.max(0, v.grade.amount)), t = 0.5 - Math.sin(Math.asin(1 - 2 * k) / 3) // inverse smoothstep
        fog.near = camera.position.distanceTo(d.base) - t * 1e4; fog.far = fog.near + 1e4
    }
    toModel([0, 0, 0], d.recv.position); d.recv.rotation.y = -yaw; d.recv.scale.setScalar(m)
    if (d.frame.visible) { d.frame.position.copy(d.recv.position); d.frame.rotation.y = -yaw; d.frame.scale.setScalar(m) }
}

// Bake the shadow textures: the case as the light sees it (cast shadow, looked
// up by every catcher) and flattened straight down (contact shadow), blurred.
function deskBake(world: any): void {
    const d = world.desk, v: DeskView = d.view, box = world.caseBox
    if (!v || !box) return
    const { THREE, renderer, scene } = world
    renderer.shadowMap.needsUpdate = true
    const y0 = box.min.y, m = world.rootScale / (v.product.scale || 1)
    const src = v.shadow.direction ?? v.lighting.key.position
    const L = new THREE.Vector3(-src[0], -src[1], -src[2]).normalize().applyAxisAngle(new THREE.Vector3(0, 1, 0), -deskYaw(d))
    if (-L.y < 0.12) { const hl = Math.hypot(L.x, L.z) || 1, k = Math.sqrt(1 - 0.12 * 0.12) / hl; L.set(L.x * k, -0.12, L.z * k) } // never quite level
    // light's-eye camera: a square round the whole case, looking along the light
    const centre = box.getCenter(new THREE.Vector3()), R = box.getSize(d.tmp).length() * 0.5 * 1.25
    const cam = d.lightCam
    cam.left = -R; cam.right = R; cam.top = R; cam.bottom = -R; cam.near = 0.01; cam.far = 2 * R + 0.02
    cam.position.copy(centre).addScaledVector(L, -(R + 0.01))
    cam.up.set(0, 1, 0); if (Math.abs(L.y) > 0.98) cam.up.set(0, 0, -1)
    cam.lookAt(centre); cam.updateProjectionMatrix(); cam.updateMatrixWorld(true)
    const u = d.shadowMat.uniforms
    u.uLight.value.set(0.5, 0, 0, 0.5, 0, 0.5, 0, 0.5, 0, 0, 0.5, 0.5, 0, 0, 0, 1).multiply(cam.projectionMatrix).multiply(cam.matrixWorldInverse)
    u.uRange.value = cam.far - cam.near
    u.uMaxLen.value = (v.shadow.maxLength ?? 1.4) * m
    u.uSoftLen.value = 0.5 * m
    u.uBase.value.copy(d.base); u.uContactBand.value = 0.01 * m
    const top = d.bakeCam
    top.position.set(d.base.x, y0 + 10, d.base.z); top.up.set(0, 0, -1); top.lookAt(d.base.x, y0, d.base.z); top.updateMatrixWorld(true)
    const hidden: any[] = []
    scene.traverse((o: any) => {
        if (o.visible && (o === world.sky || o === world.floor || o === world.ghostObject || o.isSprite || o.isPoints || o.isLine || o.isInstancedMesh)) { o.visible = false; hidden.push(o) }
    })
    const bg = scene.background, ov = scene.overrideMaterial, ac = renderer.autoClear
    const clear = renderer.getClearColor(new THREE.Color()), clearA = renderer.getClearAlpha()
    scene.background = null
    renderer.autoClear = true; renderer.setClearColor(0x000000, 0)
    scene.overrideMaterial = d.lightMat
    renderer.setRenderTarget(d.rawDir); renderer.render(scene, cam)
    scene.overrideMaterial = d.projMat
    d.projMat.uniforms.uY0.value = y0; d.projMat.uniforms.uHMax.value = 0.075
    renderer.setRenderTarget(d.rawAO); renderer.render(scene, top)
    scene.overrideMaterial = ov; scene.background = bg
    for (const o of hidden) o.visible = true
    const blur = (from: any, to: any, sx: number, sy: number) => {
        d.blurMat.uniforms.tMap.value = from.texture
        d.blurMat.uniforms.uStep.value.set(sx / to.width, sy / to.height)
        renderer.setRenderTarget(to); renderer.render(d.quadScene, d.quadCam)
    }
    // blur radii are in texels; keep them the same size on the desk whatever the case's size
    const s = Math.max(0.15, v.shadow.softness) * (0.5 / R)
    blur(d.rawDir, d.tmpA, 0.7 * s, 0); blur(d.tmpA, d.tight, 0, 0.7 * s)
    blur(d.rawDir, d.tmpB, 1.6 * s, 0); blur(d.tmpB, d.wide, 0, 1.6 * s)
    blur(d.wide, d.tmpB, 3.4 * s, 0); blur(d.tmpB, d.wide, 0, 3.4 * s)
    blur(d.wide, d.tmpB, 5.2 * s, 0); blur(d.tmpB, d.wide, 0, 5.2 * s)
    blur(d.rawAO, d.tmpC, 1.3, 0); blur(d.tmpC, d.ao, 0, 1.3)
    blur(d.ao, d.tmpC, 2.2, 0); blur(d.tmpC, d.ao, 0, 2.2)
    if (u.uGlow.value > 0) {
        blur(d.ao, d.tmpC, 4.5, 0); blur(d.tmpC, d.glow, 0, 4.5)
        blur(d.glow, d.tmpC, 8, 0); blur(d.tmpC, d.glow, 0, 8)
        // its colour: the big side covers, as lit by the key light
        const tint = u.uTint.value.set(0x000000); let n = 0
        for (const name of ["sidecover_front", "sidecover_back"]) for (const mesh of world.meshesByPart?.get(name) || []) if (mesh.material?.color) { tint.add(mesh.material.color); n++ }
        if (n) tint.multiplyScalar(1 / n); else tint.set(0x888888)
        const peak = Math.max(tint.r, tint.g, tint.b)
        if (peak > 0.02) tint.multiplyScalar(0.85 / peak) // reflected light, not paint: keep the hue, lift the level
        tint.multiply(d.tmpColor.set(v.lighting.key.color))
    }
    renderer.setRenderTarget(null)
    renderer.setClearColor(clear, clearA); renderer.autoClear = ac
}

function deskRender(world: any): void {
    const d = world.desk, { renderer, scene, camera } = world
    deskCamera(world) // also here: a resize redraws outside the animation loop
    const moving = performance.now() < d.dirtyUntil || (world.seqTasks && world.seqTasks.length > 0) || world.wobbleAnims.size > 0 || world.spawnAnims.size > 0
        || world.despawnAnims.size > 0 || world.screenholderAnims.size > 0 || world.screenRevealAnims.size > 0
    d.tick++
    if (moving) { d.settle = 2; if (!d.coarse || d.tick % 2 === 0) deskBake(world) }
    else if (d.settle > 0) { d.settle--; deskBake(world); if (d.settle === 0 && d.tex) deskBuildEnv(world, d.tex) } // one last bake (and the light map) once everything is at rest
    const ac = renderer.autoClear, bg = scene.background, skyVis = world.sky.visible
    renderer.autoClear = false
    renderer.setRenderTarget(null); renderer.clear()
    renderer.render(d.scene, camera)
    // catchers: depth first, so where two overlap on screen only the nearer one darkens the photo
    d.recvScene.overrideMaterial = d.depthMat
    renderer.render(d.recvScene, camera)
    d.recvScene.overrideMaterial = null
    renderer.render(d.recvScene, camera)
    renderer.clearDepth()
    scene.background = null; world.sky.visible = false
    renderer.render(scene, camera)
    scene.background = bg; world.sky.visible = skyVis
    if (d.grainMat.uniforms.uAmt.value > 0) {
        renderer.getDrawingBufferSize(d.bufSize)
        if (!d.frameTex || d.frameTex.image.width !== d.bufSize.x || d.frameTex.image.height !== d.bufSize.y) {
            d.frameTex?.dispose()
            d.frameTex = new world.THREE.FramebufferTexture(d.bufSize.x, d.bufSize.y)
            d.frameTex.minFilter = d.frameTex.magFilter = world.THREE.NearestFilter
            d.grainMat.uniforms.tImg.value = d.frameTex
        }
        renderer.copyFramebufferToTexture(d.zero2, d.frameTex)
        const frame = Math.floor(performance.now() / 41.7)
        if (frame !== d.grainFrame) { d.grainFrame = frame; d.grainMat.uniforms.uOff.value.set(Math.random(), Math.random(), Math.random(), Math.random()) }
        d.grainMat.uniforms.uSize.value = Math.max(1, 0.7 * renderer.getPixelRatio() + 0.4)
        renderer.render(d.grainScene, d.quadCam)
    }
    renderer.autoClear = ac
}

function deskDispose(world: any): void {
    const d = world.desk; if (!d) return
    for (const k of ["rawDir", "tight", "tmpA", "wide", "tmpB", "rawAO", "ao", "tmpC", "glow"]) d[k]?.dispose()
    for (const [, t] of d.textures) t.dispose()
    d.grainTex.dispose(); d.grainMat.dispose(); d.frameTex?.dispose()
    d.bgMat.dispose(); d.shadowMat.dispose(); d.depthMat.dispose(); d.lightMat.dispose(); d.projMat.dispose(); d.blurMat.dispose(); d.recvGeo.dispose()
    d.scene.traverse((c: any) => { c.geometry?.dispose?.() })
    world.desk = null
}
