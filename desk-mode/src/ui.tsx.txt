// ENVIRONMENT 01 / 02 / 03 and VIEW A / B, as two rows of the same pills the panels use.
function DeskSelector({ envs, env, view, onEnv, onView, accentColor, narrow }: {
    envs: DeskEnvironment[]; env: number; view: number; onEnv: (i: number) => void; onView: (i: number) => void; accentColor: string; narrow: boolean
}) {
    const row: CSSProperties = { display: "flex", alignItems: "center", gap: 6 }
    const tag: CSSProperties = { ...pixelText, fontSize: "11px", color: INK, background: "rgba(255,255,255,0.92)", borderRadius: 9999, padding: "8px 12px", whiteSpace: "nowrap" }
    const pill = (active: boolean): CSSProperties => ({ fontSize: "11px", padding: "7px 12px", minWidth: 40, ...(active ? null : { background: "rgba(255,255,255,0.92)" }) })
    return (
        <div style={{ display: "flex", flexDirection: "column", gap: 6, alignItems: "flex-start" }}>
            <div style={row} role="group" aria-label="Environment">
                <span style={tag}>{narrow ? "Env" : "Environment"}</span>
                {envs.map((e, i) => <TilePill key={e.id} active={i === env} accentColor={accentColor} onClick={() => onEnv(i)} style={pill(i === env)}>{e.id}</TilePill>)}
            </div>
            {envs[env].views.length > 1 && (
                <div style={row} role="group" aria-label="Camera view">
                    <span style={tag}>View</span>
                    {envs[env].views.map((v, i) => <TilePill key={v.id} active={i === view} accentColor={accentColor} onClick={() => onView(i)} style={pill(i === view)}>{v.id}</TilePill>)}
                </div>
            )}
        </div>
    )
}

function deskGet(o: any, path: string): any { return path.split(".").reduce((a, k) => (a == null ? a : a[k]), o) }
function deskSet(o: any, path: string, value: number): any {
    const out = JSON.parse(JSON.stringify(o)), keys = path.split(".")
    let a = out
    keys.slice(0, -1).forEach((k, i) => { if (a[k] == null) a[k] = /^\d+$/.test(keys[i + 1]) ? [] : {}; a = a[k] })
    a[keys[keys.length - 1]] = value
    return out
}
function DeskCalibrator({ view, onChange }: { view: DeskView; onChange: (v: DeskView) => void }) {
    const [copied, setCopied] = useState(false)
    const copy = () => {
        const { debugGrid, ...clean } = view as any
        const text = JSON.stringify(clean, null, 2)
        console.log("[desk view]", text)
        navigator.clipboard?.writeText(text).then(() => { setCopied(true); window.setTimeout(() => setCopied(false), 1200) }).catch(() => {})
    }
    return (
        <div style={{ position: "absolute", top: 10, left: 10, zIndex: 5, width: 232, maxHeight: "calc(100% - 20px)", overflowY: "auto", background: "rgba(255,255,255,0.94)", borderRadius: 12, padding: 10, font: "11px/1.5 ui-monospace, monospace", color: INK }}>
            <div style={{ display: "flex", gap: 6, marginBottom: 6 }}>
                <button onClick={() => onChange({ ...view, debugGrid: !view.debugGrid })}>{view.debugGrid ? "Hide grid" : "Show grid"}</button>
                <button onClick={copy}>{copied ? "Copied" : "Copy JSON"}</button>
            </div>
            {DESK_CAL_FIELDS.map(([path, step]) => (
                <label key={path} style={{ display: "flex", justifyContent: "space-between", alignItems: "center", gap: 6 }}>
                    <span>{path}</span>
                    <input type="number" step={step} value={deskGet(view, path) ?? ""} style={{ width: 70, font: "inherit" }}
                        onChange={(e) => { const n = parseFloat(e.target.value); if (Number.isFinite(n)) onChange(deskSet(view, path, n)) }} />
                </label>
            ))}
        </div>
    )
}
