// ── Desk mode: photographic environments ─────────────────────────────────────
// Each view is ONE static photograph ("plate") with the live case composited
// into it. Nothing here is a 3D room: a view only describes the camera that
// took the photo, where the case stands on the desk, the light that hits it
// and the shadow it throws. Everything is in "desk space": metres, y up, the
// desk top at y = 0. Tune a view by opening the configurator with ?deskcal
// in the URL (live number fields + a floor grid; copy the JSON back in here).
type DeskV3 = [number, number, number]
interface DeskLight { color: number; intensity: number; position: DeskV3 } // position = direction the light comes FROM
interface DeskView {
    id: string; label: string
    // the photograph. srcSmall is used on small screens. width/height = the full image's pixels (aspect only).
    image: { src: string; srcSmall?: string; width: number; height: number }
    // the camera that took it. fov = vertical field of view of the WHOLE image, degrees. roll in degrees.
    camera: { position: DeskV3; target: DeskV3; fov: number; roll?: number }
    // where the case stands: [x, z] on the desk, turned rotationY degrees, scale 1 = true size.
    product: { position: [number, number]; rotationY: number; scale: number }
    // how the photo is cropped to the viewer: focus = the point (0..1 across, 0..1 down) kept in view,
    // zoom / narrowZoom (phones) crop in tighter than plain "cover".
    framing?: { focus?: [number, number]; zoom?: number; narrowZoom?: number }
    lighting: {
        ambient: { color: number; intensity: number }
        key: DeskLight; fill: DeskLight; rim: DeskLight
        bounce?: { sky: number; ground: number; intensity: number } // light off the wall (sky) and the desk (ground)
    }
    // shadow catcher: colour + strength of the cast shadow, how soft it gets away from the case,
    // strength of the contact shadow under it. direction overrides the key light's.
    // glow = how much of its own colour the case throws back onto the desk round its foot (0 = none).
    shadow: { color: number; opacity: number; softness: number; contact: number; direction?: DeskV3; glow?: number }
    portrait?: Partial<DeskView> // overrides for tall (phone) viewers, e.g. a separate portrait photograph
    debugGrid?: boolean
}
interface DeskEnvironment { id: string; name: string; views: DeskView[] }
